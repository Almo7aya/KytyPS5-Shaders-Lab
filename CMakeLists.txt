cmake_minimum_required(VERSION 3.24)
project(PS5ShaderLab VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SHADER_LAB_BUILD_KYTY_WORKER "Build a worker against a user-selected KytyPS5 checkout" OFF)
set(KYTY_ROOT "" CACHE PATH "KytyPS5 checkout (never modified)")
set(SHADER_LAB_DEPS "${KYTY_ROOT}/3rdparty" CACHE PATH "Optional existing nlohmann_json and xxHash source directory")
include(FetchContent)
if(EXISTS "${SHADER_LAB_DEPS}/nlohmann_json/include/nlohmann/json.hpp")
  set(SL_JSON_INCLUDE "${SHADER_LAB_DEPS}/nlohmann_json/include")
else()
  FetchContent_Declare(sl_json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.12.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(sl_json)
  set(SL_JSON_INCLUDE "${sl_json_SOURCE_DIR}/include")
endif()
if(EXISTS "${SHADER_LAB_DEPS}/xxHash/xxhash.h")
  set(SL_XXHASH_INCLUDE "${SHADER_LAB_DEPS}/xxHash")
else()
  FetchContent_Declare(sl_xxhash GIT_REPOSITORY https://github.com/Cyan4973/xxHash.git GIT_TAG v0.8.3 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(sl_xxhash)
  set(SL_XXHASH_INCLUDE "${sl_xxhash_SOURCE_DIR}")
endif()
find_package(Threads REQUIRED)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(sl_zlib GIT_REPOSITORY https://github.com/madler/zlib.git GIT_TAG v1.3.2 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(sl_zlib)
add_library(shader_lab_core src/core.cpp src/scan.cpp src/containers.cpp src/process.cpp src/run.cpp src/campaign.cpp src/reference.cpp src/execution.cpp src/repro.cpp src/minimize.cpp src/pass_catalog.cpp src/pass_bisect.cpp src/capture.cpp src/pixel_profile.cpp src/host_profile.cpp src/analysis.cpp src/report.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/scan.cpp src/containers.cpp src/core.cpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/scan.cpp" SL_SCAN_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/containers.cpp" SL_CONTAINER_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/core.cpp" SL_CORE_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" SL_BUILD_HASH)
string(SHA256 SL_EXTRACTOR_ID "${SL_SCAN_HASH}${SL_CONTAINER_HASH}${SL_CORE_HASH}${SL_BUILD_HASH}")
target_compile_definitions(shader_lab_core PRIVATE "SL_EXTRACTOR_ID=\"${SL_EXTRACTOR_ID}\"")
target_include_directories(shader_lab_core PUBLIC include "${SL_JSON_INCLUDE}" PRIVATE "${SL_XXHASH_INCLUDE}")
target_link_libraries(shader_lab_core PUBLIC Threads::Threads)
target_include_directories(shader_lab_core PRIVATE "${sl_zlib_SOURCE_DIR}" "${sl_zlib_BINARY_DIR}")
target_link_libraries(shader_lab_core PRIVATE zlibstatic)
if(WIN32)
  target_compile_definitions(shader_lab_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(shader_lab_core PRIVATE bcrypt)
endif()
if(MSVC)
  target_compile_options(shader_lab_core PRIVATE /W4 /EHsc)
else()
  target_compile_options(shader_lab_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(shader-lab src/main.cpp)
target_link_libraries(shader-lab PRIVATE shader_lab_core)
add_library(shader_lab_cpu_model STATIC src/cpu_reference.cpp)
target_link_libraries(shader_lab_cpu_model PUBLIC shader_lab_core)
add_executable(shader-cpu-reference src/cpu_worker.cpp)
target_link_libraries(shader-cpu-reference PRIVATE shader_lab_cpu_model)
enable_testing()
add_executable(shader-lab-tests tests/tests.cpp tests/reference_tests.cpp tests/execution_tests.cpp tests/cpu_tests.cpp tests/archive_tests.cpp tests/repro_tests.cpp tests/minimize_tests.cpp tests/pass_tests.cpp tests/capture_tests.cpp tests/pixel_tests.cpp tests/host_tests.cpp)
target_link_libraries(shader-lab-tests PRIVATE shader_lab_core)
target_compile_definitions(shader-lab-tests PRIVATE "SL_TEST_SOURCE_DIR=\"${CMAKE_CURRENT_SOURCE_DIR}/tests\"")
target_include_directories(shader-lab-tests PRIVATE "${sl_zlib_SOURCE_DIR}" "${sl_zlib_BINARY_DIR}")
target_link_libraries(shader-lab-tests PRIVATE zlibstatic)
if(MSVC)
  target_compile_options(shader_lab_cpu_model PRIVATE /W4 /EHsc)
  target_compile_options(shader-cpu-reference PRIVATE /EHsc)
  target_sources(shader-lab PRIVATE cmake/windows.manifest)
  target_sources(shader-lab-tests PRIVATE cmake/windows.manifest)
  target_sources(shader-cpu-reference PRIVATE cmake/windows.manifest)
endif()
add_dependencies(shader-lab-tests shader-cpu-reference)
add_test(NAME shader_lab_unit COMMAND shader-lab-tests)
add_test(NAME shader_lab_help COMMAND shader-lab --help)
add_test(NAME shader_lab_cpu_reference COMMAND shader-lab-tests --cpu-worker $<TARGET_FILE:shader-cpu-reference>)
if(SHADER_LAB_BUILD_KYTY_WORKER)
  include(cmake/KytyWorker.cmake)
  add_test(NAME shader_lab_real_kyty COMMAND shader-lab-tests --real-worker $<TARGET_FILE:shader_cfg_tests>)
endif()
